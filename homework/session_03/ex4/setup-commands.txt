# Create web roots
sudo mkdir -p /var/www/beta-app/html
sudo mkdir -p /var/www/internal-app/html

# Copy website files
sudo cp beta-app/index.html /var/www/beta-app/html/index.html
sudo cp internal-app/index.html /var/www/internal-app/html/index.html

# Copy and enable Nginx configuration
sudo cp multi-port.conf /etc/nginx/sites-available/multi-port.conf
sudo ln -s /etc/nginx/sites-available/multi-port.conf /etc/nginx/sites-enabled/multi-port.conf
sudo nginx -t
sudo systemctl reload nginx

# Open UFW ports
sudo ufw allow 8080/tcp
sudo ufw allow 8090/tcp
sudo ufw status

# DigitalOcean Cloud Firewall
# Add inbound TCP rules for ports 8080 and 8090 in the DigitalOcean dashboard.
